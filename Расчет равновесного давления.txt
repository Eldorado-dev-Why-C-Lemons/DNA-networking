def calc_z_factor(P_mpa, T_K, Tc, Pc, omega):
  R = 8.31446
  P = float(P_mpa) * 1e6
  Pc_pa = float(Pc) * 1e6

  Tr = float(T_K) / float(Tc)
  kappa = 0.37464 + 1.54226 * omega - 0.26992 * (omega**2)
  alpha = (1.0 + kappa * (1.0 - np.sqrt(Tr))) ** 2

  a = 0.45724 * (R**2) * (Tc**2) / Pc_pa * alpha
  b = 0.07780 * R * Tc / Pc_pa

  A = a * P / ((R * T_K) ** 2)
  B = b * P / (R * T_K)

  coeffs = [
      1.0,
      -(1.0 - B),
      (A - 3.0 * (B**2) - 2.0 * B),
      -(A * B - (B**2) - (B**3)),
  ]
  roots = np.roots(coeffs)
  real_roots = roots[np.isreal(roots)].real
  return float(np.max(real_roots))
