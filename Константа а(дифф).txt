import numpy as np

# @title Выберите газ и температуру (°C) { run: "auto" }
gas_name = "Углекислый газ (CO2)"  # @param ["Метан (CH4)", "Этан (C2H6)", "Пропан (C3H8)", "Углекислый газ (CO2)", "Сероводород (H2S)"]
T_celsius = 3.0  # @param {type:"number"}


def estimate_alpha_diff(gas_name, T_celsius):
  T_K = float(T_celsius) + 273.15
  T_ref = 276.15  # 3 °C
  R = 8.314

  # Справочная база фундаментальных молекулярных свойств:
  # d_nm: кинетический диаметр молекулы (нм)
  # H_298: константа Генри при 25 °C, моль/(м³·Па)
  # dH_sol: энтальпия растворения (-dH/R, К)
  # N0: молярная плотность газа в гидрате, моль/м³
  db = {
      "Метан (CH4)": {
          "d_nm": 0.380,
          "H_298": 1.4e-5,
          "dH_R": 1600.0,
          "N0": 7500.0,
      },
      "Этан (C2H6)": {
          "d_nm": 0.444,
          "H_298": 1.9e-5,
          "dH_R": 2400.0,
          "N0": 7200.0,
      },
      "Пропан (C3H8)": {
          "d_nm": 0.510,
          "H_298": 1.5e-5,
          "dH_R": 2700.0,
          "N0": 2900.0,
      },
      "Углекислый газ (CO2)": {
          "d_nm": 0.330,
          "H_298": 3.4e-4,
          "dH_R": 2400.0,
          "N0": 7400.0,
      },
      "Сероводород (H2S)": {
          "d_nm": 0.360,
          "H_298": 1.0e-3,
          "dH_R": 2100.0,
          "N0": 7600.0,
      },
  }

  prop = db[gas_name]
  d_ch4 = db["Метан (CH4)"]["d_nm"]

  # 1. Расчет константы Генри H(T) при заданной температуре, моль/(м³·Па)
  H_T = prop["H_298"] * np.exp(prop["dH_R"] * (1.0 / T_K - 1.0 / 298.15))

  # 2. Эффективный коэффициент диффузии в пористой корке D_eff(T), м²/с
  # Калибровка по эталонному D_eff для метана при 3 °C (~5.8e-12 м²/с)
  D_ch4_ref = 5.8e-12
  E_diff = 55000.0
  size_factor = (d_ch4 / prop["d_nm"]) ** 2
  temp_factor = np.exp((-E_diff / R) * (1.0 / T_K - 1.0 / T_ref))

  D_eff = D_ch4_ref * size_factor * temp_factor

  # 3. Итоговый коэффициент alpha_diff, м²/(с·МПа)
  alpha_diff = (D_eff * H_T / prop["N0"]) * 1e6

  print(f"=== ОЦЕНКА ПАРАМЕТРОВ ПЕРЕНОСА: {gas_name} ПРИ {T_celsius} °C ===")
  print(f"Диаметр молекулы (d):          {prop['d_nm']} нм")
  print(f"Растворимость Генри H(T):      {H_T:.3e} моль/(м³·Па)")
  print(f"Эфф. диффузия в корке D_eff:   {D_eff:.3e} м²/с")
  print(f"Плотность упаковки N0:         {prop['N0']:.0f} моль/м³")
  print("-" * 55)
  print(f"-> РАСЧЕТНЫЙ alpha_diff:       {alpha_diff:.3e} м²/(с·МПа)")


estimate_alpha_diff(gas_name, T_celsius)
